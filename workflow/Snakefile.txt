## raw_read_QC: Perform quality control using FastQC (Step 2)
# Step 2: Raw read QC
# Original Slurm script: slurm_fastqc.sh (array = 1-48)
rule raw_read_QC:
    input:
        r1 = config["input_dir"] + SAMPLE_NAME + "{num}_R1.fastq.gz"
        r2 = config["input_dir"] + SAMPLE_NAME + "{num}_R2.fastq.gz"
    output:
        html = OUTDIR + "fastqc/" + SAMPLE_NAME + "{num}_R1_fastq.html"
        zip =  OUTDIR + "fastqc/" + SAMPLE_NAME + "{num}_R1_fastq.zip"
    params:
        outdir = OUTDIR
    envmodules:
        "fastqc/0.11.9"
    shell:
        "fastqc -t 2 {input.r1} {input.r2} -o {params.outdir}"
