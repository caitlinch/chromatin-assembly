## mask_repeats: Masking with RepeatMasker (Step 1a)
# Step 1: Repeat masking
# Original Slurm script: slurm_waterdragon_RM.sh
rule mask_repeats:
    input:
        fasta = config["input"]["reference_genome_dir"] + config["input"]["reference_genome_file"]
    output:
        masked = OUTDIR + config["input"]["reference_genome_file"] + ".masked"
    params:
        adapter = config["params"]["adapters"]
        outdir = OUTDIR
    envmodules:
        "repeatmasker/4.1.0"
    shell:
        """
        RepeatMasker -pa config["step1"]["slurm_job_cpus"] -qq -dir {params.outdir} -lib {params.adapter} {input.fasta}
        """
