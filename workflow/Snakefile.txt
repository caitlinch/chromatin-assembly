## raw_read_QC: Perform quality control using FastQC (Step 2)
# Step 2: Raw read QC
rule raw_read_QC:
    input:
        r1 = config["input"]["input_dir"] + SAMPLE_NAME + "{num}_R1.fastq.gz",
        r2 = config["input"]["input_dir"] + SAMPLE_NAME + "{num}_R2.fastq.gz"
    output:
        zip1 =  OUTDIR + "fastqc/" + SAMPLE_NAME + "{num}_R1_fastq.zip",
        zip2 =  OUTDIR + "fastqc/" + SAMPLE_NAME + "{num}_R2_fastq.zip"
    params:
        outdir = OUTDIR
    shell:
        """
        module load fastqc/0.11.9
        touch {output.zip1} | touch {output.zip2}
        """
